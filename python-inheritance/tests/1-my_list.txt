>>> from 1-my_list import MyList
>>> ml = MyList()
>>> ml.append(4)
>>> ml.append(1)
>>> ml.append(3)
>>> ml.append(2)
>>> ml
[4, 1, 3, 2]

# print_sorted should print the sorted list but NOT modify the original
>>> ml.print_sorted()
[1, 2, 3, 4]

# After printing sorted, original list must stay the same
>>> ml
[4, 1, 3, 2]

# Adding more elements still works because MyList inherits from list
>>> ml.append(5)
>>> ml
[4, 1, 3, 2, 5]
